// Disimpan sebagai .txt agar tidak ikut typecheck. Salin ke src/i18n/request.ts bila disetujui.
import { getRequestConfig } from "next-intl/server";
import { routing } from "./routing";

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale;
  const locale = routing.locales.find((l) => l === requested) ?? routing.defaultLocale;
  return {
    locale,
    // Zona waktu tetap: hasil render saat build dan di peramban sama, tanggal tidak bergeser.
    timeZone: "Asia/Jakarta",
    messages: (await import(`../../messages/${locale}.json`)).default,
  };
});
