import type { LikertConfig } from "@/types";

/**
 * Skor akhir sebuah jawaban Likert dengan reverse scoring. Fungsi murni.
 * Mengembalikan null bila nilai tidak valid (opsi kosong, atau nilai bukan salah satu opsi),
 * agar pemanggil tidak menghitung rata-rata dari NaN.
 * Catatan: `direction` hanya mengatur urutan tampilan; skor mengikuti `value` opsi.
 */
export function scoreLikert(value: number, config: LikertConfig): number | null {
  const values = config.options.map((o) => o.value);
  if (values.length === 0 || !values.includes(value)) return null;
  if (!config.reverseScored) return value;
  return Math.max(...values) + Math.min(...values) - value;
}
